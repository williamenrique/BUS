-- =================================================================
-- SCRIPT DE MIGRACIÓN DE TIPOS DE DATOS DE FECHA Y HORA
-- Objetivo: Convertir columnas VARCHAR a DATE, TIME o DATETIME
-- sin pérdida de datos.
-- =================================================================

-- Desactivar temporalmente el modo seguro para permitir actualizaciones sin WHERE en la clave primaria.
SET SQL_SAFE_UPDATES = 0;

-- -----------------------------------------------------------------
-- Tabla: table_alm_despacho
-- Columna: fecha_despacho (VARCHAR -> DATE)
-- -----------------------------------------------------------------
ALTER TABLE `table_alm_despacho` ADD COLUMN `fecha_despacho_temp` DATE NULL AFTER `fecha_despacho`;

UPDATE `table_alm_despacho`
SET `fecha_despacho_temp` = COALESCE(
    STR_TO_DATE(TRIM(`fecha_despacho`), '%Y-%m-%d'),
    STR_TO_DATE(TRIM(`fecha_despacho`), '%d-%m-%Y')
);

ALTER TABLE `table_alm_despacho` DROP COLUMN `fecha_despacho`;
ALTER TABLE `table_alm_despacho` CHANGE COLUMN `fecha_despacho_temp` `fecha_despacho` DATE NULL;

-- -----------------------------------------------------------------
-- Tabla: table_es_cierre
-- Columna: fecha_cierre (VARCHAR -> DATE)
-- Columna: hora_cierre (VARCHAR -> TIME)
-- -----------------------------------------------------------------
ALTER TABLE `table_es_cierre` ADD COLUMN `fecha_cierre_temp` DATE NULL AFTER `fecha_cierre`;
ALTER TABLE `table_es_cierre` ADD COLUMN `hora_cierre_temp` TIME NULL AFTER `hora_cierre`;

UPDATE `table_es_cierre`
SET 
    `fecha_cierre_temp` = COALESCE(
        STR_TO_DATE(TRIM(`fecha_cierre`), '%Y-%m-%d'),
        STR_TO_DATE(TRIM(`fecha_cierre`), '%d-%m-%Y')
    ),
    `hora_cierre_temp` = STR_TO_DATE(TRIM(`hora_cierre`), '%H:%i:%s');

ALTER TABLE `table_es_cierre` DROP COLUMN `fecha_cierre`;
ALTER TABLE `table_es_cierre` DROP COLUMN `hora_cierre`;
ALTER TABLE `table_es_cierre` CHANGE COLUMN `fecha_cierre_temp` `fecha_cierre` DATE NULL;
ALTER TABLE `table_es_cierre` CHANGE COLUMN `hora_cierre_temp` `hora_cierre` TIME NULL;

        STR_TO_DATE(TRIM(`fecha_entrada`), '%Y-%m-%d'),

    ),
    `fecha_salida_temp` = COALESCE(
        STR_TO_DATE(TRIM(`fecha_salida`), '%Y-%m-%d %H:%i:%s'),
